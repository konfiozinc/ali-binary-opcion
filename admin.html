const SHEET_NAME = 'Signals'; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const sheet = getSheet(); if (data.type === 'signal') { // Guardar señal - borrar señal activa anterior primero clearActiveSignal(sheet); sheet.appendRow([ new Date().toISOString(), data.asset, data.direction, data.broker, 'SIGNAL', '', Date.now().toString() ]); } if (data.type === 'result') { // Buscar la última señal activa y agregarle el resultado const lastRow = findLastSignalRow(sheet); if (lastRow > 0) { sheet.getRange(lastRow, 6).setValue(data.result); sheet.getRange(lastRow, 5).setValue('RESULT'); } } return ContentService .createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ ok: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { const sheet = getSheet(); const lastRow = sheet.getLastRow(); if (lastRow < 2) { return jsonResponse({ signal: null }); } // Leer última fila const row = sheet.getRange(lastRow, 1, 1, 7).getValues()[0]; const response = { signalId: row[6] ? row[6].toString() : lastRow.toString(), timestamp: row[0], asset: row[1], direction: row[2], broker: row[3], status: row[4], result: row[5] || null, resultId: row[5] ? (row[6] + '_r') : null }; return jsonResponse(response); } catch(err) { return jsonResponse({ signal: null, error: err.toString() }); } } function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(['Timestamp','Asset','Direction','Broker','Status','Result','ID']); } return sheet; } function clearActiveSignal(sheet) { // No borramos, solo marcamos } function findLastSignalRow(sheet) { const lastRow = sheet.getLastRow(); if (lastRow < 2) return 0; return lastRow; } function jsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
